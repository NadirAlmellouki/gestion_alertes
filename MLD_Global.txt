Table organizational_unit {

  id uuid [pk]

  code varchar(100) [unique]

  name varchar(255) [not null]

  unit_type varchar(20) [
    not null,
    note: 'CHECK (ENTITY, POLE, DOMAIN, SOLUTION) — réellement appliqué en SQL'
  ]

  parent_unit_id uuid

  is_subsidiary boolean [default:false, note: 'Colonne "Filiale" du référentiel Entités']

  active boolean [default:true]

  created_at timestamptz

  updated_at timestamptz
}

Table solution_attribute {

  unit_id uuid [pk]

  solution_type varchar(100) [note: 'CHECK (Progiciel, InHouse)']

  psi varchar(100) [
    note: 'CHECK (P0,P1,P2,P3,P4) — PRIORITE METIER FIXE DE LA SOLUTION (Master Prompt V2). Jamais calculee par le LLM, jamais modifiee par le moteur de regles ni le moteur de routage. Ne pas confondre avec business_rule.rule_evaluation_priority (ordre d evaluation d une regle, notion totalement differente).'
  ]

  service_type varchar(100) [note: 'CHECK (Applicatif, Technique)']

  tenant varchar(100)

  target_scope varchar(150) [note: 'anciennement "popularity" (integer, errone) — colonne Excel "Popularite", categorielle et multi-valeurs']

  functional_description text [note: 'colonne Excel "Role"']

  active boolean
}

////////////////////////////////////////////////////////////
//
// PERSON
//
// Person est l'unique concept metier representant un individu.
// "Administrator" / "Supervisor" / "Operator" ne sont PAS des entites
// de domaine separees : ce sont des roles Keycloak (autorisation
// applicative), geres exclusivement par Keycloak, jamais stockes ici.
// Le role METIER d'une Person (TAM, DBA, Infra, Responsable...) est
// porte par unit_admin_assignment.role, totalement independant de son
// eventuel role Keycloak. Voir Master Prompt V2, section ROLES : KEYCLOAK != ROLE METIER.
//
////////////////////////////////////////////////////////////

Table person {

  id uuid [pk]

  first_name varchar(150)

  last_name varchar(150)

  full_name varchar(255) [not null]

  email varchar(255) [note: 'UNIQUE INDEX partiel sur lower(email) — cle naturelle d idempotence pour l import Excel']

  phone varchar(100)

  active boolean [default:true]

  created_at timestamptz

  updated_at timestamptz
}

// Referentiel technique des roles METIER (remplace le CHECK fige a 4 valeurs :
// TAM/TECHNICAL/FUNCTIONAL/RECETTE, qui ne couvrait que 4 des 12 roles reels du
// referentiel Excel — 11 colonnes de role de la feuille Solutions + le role
// "Responsable" porte par chaque Entite/Pole/Domaine). Donnee technique fixe,
// independante des roles Keycloak.
Table admin_role_type {

  code varchar(40) [pk]

  label varchar(150) [not null]

  excel_column varchar(150)

  applicable_scope varchar(20) [
    not null,
    default: 'SOLUTION',
    note: 'CHECK (UNIT, SOLUTION) — UNIT = Entite/Pole/Domaine (role Responsable), SOLUTION = uniquement au niveau Solution'
  ]

  display_order integer

  active boolean [default:true]
}

Table unit_admin_assignment {

  id uuid [pk]

  unit_id uuid

  person_id uuid

  role varchar(40) [
    not null,
    note: 'FK -> admin_role_type.code (role METIER, ex: TAM/DBA/INFRA_ADMIN/MANAGER... — sans rapport avec un role Keycloak)'
  ]

  primary_contact boolean

  active boolean

  created_at timestamptz

  Indexes {
    (unit_id, person_id, role) [unique, note: 'empeche tout doublon d affectation ; permet un upsert idempotent depuis l Excel']
  }
}

Table member_availability {

  id uuid [pk]

    note: 'CHECK(AVAILABLE,BUSY,VACATION,OFFLINE) — reellement applique en SQL'

  valid_to timestamptz
}

////////////////////////////////////////////////////////////
//
// BUSINESS RULE ENGINE
//
// Business Rule
//     |
//     +-- Condition Groups (rule_condition_group, block_type = CONDITION)
//     |       +-- rule_condition (operateur AND/OR porte par le groupe)
//     |
//     +-- Exceptions (rule_condition_group, block_type = EXCEPTION)
//     |       +-- rule_condition (meme modele que les conditions)
//     |
//     +-- Actions (rule_action) — ne modifient JAMAIS Solution/Domaine/Pole/
//         Entite/PSI (voir CHECK chk_rule_action_type_not_referential)
//
// Une regle s'applique si ses groupes CONDITION sont satisfaits ET qu'aucun de
// ses groupes EXCEPTION n'est vrai.
//
////////////////////////////////////////////////////////////

Table business_rule {

  id uuid [pk]

  code varchar(50) [unique]

  name varchar(255)

  description text

  rule_evaluation_priority integer [
    note: 'Ordre d evaluation de CETTE regle uniquement (valeur basse = evaluee en premier). SANS RAPPORT avec solution_attribute.psi — ne jamais confondre (Master Prompt V2).'
  ]

  enabled boolean

  stop_on_match boolean [
    note: 'Si vrai et que la regle correspond, arrete l evaluation des regles suivantes. Mecanisme unique d arret en V0 (pas de FIRST_MATCH/ALL_MATCHES global separe).'
  ]

  created_at timestamptz
}

Table rule_condition_group {

  id uuid [pk]

  rule_id uuid

  block_type varchar(10) [
    default: 'CONDITION',
    note: 'CHECK (CONDITION, EXCEPTION) — un meme groupe sert soit de bloc de conditions, soit de bloc d exceptions de la regle'
  ]

  logical_operator varchar(5) [note: 'CHECK (AND, OR)']

  execution_order integer
}

Table rule_condition {

  id uuid [pk]

  group_id uuid

  field_name varchar(150)

  operator varchar(30)

  expected_value text

  value_type varchar(30)

  condition_order integer
}

Table rule_action {

  ]

  action_value text

  execution_order integer
}

Table rule_execution {

Table routing_step {

  id uuid [pk]

  routing_policy_id uuid

  step_order integer [note: 'ordre de la sequence — le comportement du moteur est porte par cet ordre, pas par des drapeaux continue_if_*']

  delay_after_seconds integer [note: 'delai propre a CETTE etape avant la suivante — plus de delai/nombre de relances global']
